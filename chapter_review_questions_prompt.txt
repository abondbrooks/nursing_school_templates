        // Fisher-Yates Shuffle function

        function shuffleArray(array) {

            for (let i = array.length - 1; i > 0; i--) {

                const j = Math.floor(Math.random() * (i + 1));

                [array[i], array[j]] = [array[j], array[i]];

            }

        }



        let currentQuestionIndex = 0;

        let score = 0;

        let missedQuestions = [];



        const quizContainer = document.getElementById('quiz-container');

        const questionCounterEl = document.getElementById('question-counter');

        const questionTextEl = document.getElementById('question-text');

        const optionsContainerEl = document.getElementById('options-container');

        const feedbackEl = document.getElementById('feedback');

        const feedbackTitleEl = document.getElementById('feedback-title');

        const feedbackRationaleEl = document.getElementById('feedback-rationale');

        const feedbackSourceEl = document.getElementById('feedback-source');

        const testTipContainer = document.getElementById('test-tip-container');

        const submitBtn = document.getElementById('submit-btn');

        const nextBtn = document.getElementById('next-btn');

        const resultsContainer = document.getElementById('results-container');

        const scoreCircleEl = document.getElementById('score-circle');

        const resultMessageEl = document.getElementById('result-message');

        const retakeAllBtn = document.getElementById('retake-all-btn');

        const retakeMissedBtn = document.getElementById('retake-missed-btn');

        const progressFillEl = document.getElementById('progress-fill');



        // Initial Shuffle

        shuffleArray(quizData);



        function loadQuestion() {

            const currentQuestion = quizData[currentQuestionIndex];

            const isMulti = currentQuestion.type === 'multi';



            // Update progress

            const progress = ((currentQuestionIndex) / quizData.length) * 100;

            progressFillEl.style.width = `${progress}%`;



            questionCounterEl.textContent = `Question ${currentQuestionIndex + 1} of ${quizData.length}`;

            questionTextEl.textContent = currentQuestion.question;



            optionsContainerEl.innerHTML = '';



            if(isMulti) {

                const hint = document.createElement('span');

                hint.className = 'sata-instruction';

                hint.textContent = "Select All That Apply";

                optionsContainerEl.appendChild(hint);

            }



            currentQuestion.options.forEach((option, index) => {

                const optionItem = document.createElement('div');

                optionItem.className = 'option-item';



                const inputType = isMulti ? 'checkbox' : 'radio';

                const inputId = `option${index}`;



                optionItem.innerHTML = `

                    <label for="${inputId}" class="option-label">

                        <input type="${inputType}" id="${inputId}" name="option" value="${index}">

                        <span>${option}</span>

                    </label>

                `;

                optionsContainerEl.appendChild(optionItem);

            });



            // Add listeners

            const inputs = optionsContainerEl.querySelectorAll('input');

            inputs.forEach(input => {

                input.addEventListener('change', () => {

                    // Visual selection style

                    if (!isMulti) {

                        // clear all selected styles

                        optionsContainerEl.querySelectorAll('.option-label').forEach(l => l.classList.remove('selected'));

                    }



                    if (input.checked) {

                        input.parentElement.classList.add('selected');

                    } else {

                        input.parentElement.classList.remove('selected');

                    }



                    // Enable submit button if at least one checked

                    const anyChecked = Array.from(inputs).some(i => i.checked);

                    submitBtn.disabled = !anyChecked;

                });

            });



            feedbackEl.classList.add('hidden');

            submitBtn.classList.remove('hidden');

            submitBtn.disabled = true;

            nextBtn.classList.add('hidden');

        }



        function checkAnswer() {

            const currentQuestion = quizData[currentQuestionIndex];

            const isMulti = currentQuestion.type === 'multi';

            const inputs = optionsContainerEl.querySelectorAll('input');



            let isCorrect = false;

            let userSelectedIndices = [];



            inputs.forEach(input => {

                if (input.checked) userSelectedIndices.push(parseInt(input.value));

                input.disabled = true; // Disable inputs

            });



            if (isMulti) {

                const correctIndices = currentQuestion.answer.sort();

                userSelectedIndices.sort();

                isCorrect = (correctIndices.length === userSelectedIndices.length) &&

                            correctIndices.every((val, index) => val === userSelectedIndices[index]);

            } else {

                isCorrect = userSelectedIndices[0] === currentQuestion.answer;

            }



            // Styling Feedback

            const labels = optionsContainerEl.querySelectorAll('.option-label');

            labels.forEach((label, index) => {

                label.classList.remove('selected'); // Remove selection blue



                const isAnswerCorrect = isMulti ? currentQuestion.answer.includes(index) : index === currentQuestion.answer;

                const wasSelected = userSelectedIndices.includes(index);



                if (isAnswerCorrect) {

                    label.classList.add('correct-answer'); // Green

                    if (!wasSelected) label.classList.add('missed-answer');

                } else if (wasSelected) {

                    label.classList.add('wrong-answer'); // Red

                }

            });



            if (isCorrect) {

                score++;

                feedbackTitleEl.textContent = "Correct!";

                feedbackTitleEl.className = "feedback-correct";

            } else {

                missedQuestions.push(currentQuestion);

                feedbackTitleEl.textContent = "Incorrect";

                feedbackTitleEl.className = "feedback-incorrect";

            }



            feedbackRationaleEl.textContent = currentQuestion.rationale;

            feedbackSourceEl.textContent = `Source: ${currentQuestion.source}`;

            

            // Test Tip

            testTipContainer.innerHTML = '';

            if(currentQuestion.tip) {

                const tipDiv = document.createElement('div');

                tipDiv.className = 'test-tip';

                tipDiv.textContent = `💡 Exam Tip: ${currentQuestion.tip}`;

                testTipContainer.appendChild(tipDiv);

            }



            feedbackEl.classList.remove('hidden');

            submitBtn.classList.add('hidden');

            nextBtn.classList.remove('hidden');



            // Scroll to feedback

            feedbackEl.scrollIntoView({ behavior: 'smooth' });

        }



        function nextQuestion() {

            currentQuestionIndex++;

            if (currentQuestionIndex < quizData.length) {

                loadQuestion();

                quizContainer.scrollIntoView({ behavior: 'smooth', block: 'start' });

            } else {

                showResults();

            }

        }



        function showResults() {

            quizContainer.querySelector('#question-area').classList.add('hidden');

            feedbackEl.classList.add('hidden');

            document.getElementById('navigation').classList.add('hidden');



            const percentage = Math.round((score / quizData.length) * 100);

            scoreCircleEl.textContent = `${percentage}%`;



            let msg = "";

            if (percentage >= 80) msg = "Excellent work! You are ready for the Chapter 14 exam.";

            else if (percentage >= 60) msg = "Good job, but review the missed concepts in the textbook.";

            else msg = "Keep studying. Focus on the anatomy and blood flow sections.";



            resultMessageEl.textContent = msg;

            resultsContainer.classList.remove('hidden');



            if (missedQuestions.length === 0) {

                retakeMissedBtn.classList.add('hidden');

            } else {

                retakeMissedBtn.classList.remove('hidden');

            }

        }



        submitBtn.addEventListener('click', checkAnswer);

        nextBtn.addEventListener('click', nextQuestion);



        retakeAllBtn.addEventListener('click', () => {

            location.reload();

        });



        retakeMissedBtn.addEventListener('click', () => {

            if (missedQuestions.length > 0) {

                quizData.length = 0;

                missedQuestions.forEach(q => quizData.push(q));

                

                // Shuffle missed

                shuffleArray(quizData);



                currentQuestionIndex = 0;

                score = 0;

                missedQuestions = [];



                resultsContainer.classList.add('hidden');

                document.getElementById('question-area').classList.remove('hidden');

                document.getElementById('navigation').classList.remove('hidden');

                loadQuestion();

            }

        });



        // Initialize

        loadQuestion();
